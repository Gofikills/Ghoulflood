# systems/empire.py # Player's cat empire: level, rooms, passive income, upgrades. # Casino is the income source, empire is the long-term goal. class Empire: def __init__(self): self.level = 1 self.xp = 0 self.xp_to_next = 100 # Simple rooms that can be upgraded later self.rooms = { "lobby": { "level": 1, "name": "Лобби", "bonus": 0 }, "slots_hall": { "level": 1, "name": "Зал слотов", "bonus": 0 }, "vip": { "level": 0, "name": "VIP-зона", "bonus": 0 }, } self.passive_income_rate = 0 self.last_passive_claim = 0 def add_xp(self, amount: int): self.xp += amount while self.xp >= self.xp_to_next: self.xp -= self.xp_to_next self.level += 1 self.xp_to_next = int( self.xp_to_next * 1.5 ) # Unlock things at levels if self.level == 2: self.rooms["vip"]["level"] = 1 self.passive_income_rate = self.level * 1.5 def get_passive_income(self, collection) -> int: base = self.passive_income_rate # Cats with passive ability abilities = collection.get_active_abilities() extra = abilities.get("passive_income", 0) return int(base + extra) def to_dict(self) -> dict: return { "level": self.level, "xp": self.xp, "xp_to_next": self.xp_to_next, "rooms": self.rooms, "passive_income_rate": self.passive_income_rate, } def load_dict(self, data: dict): self.level = data.get("level", 1) self.xp = data.get("xp", 0) self.xp_to_next = data.get("xp_to_next", 100) self.rooms = data.get("rooms", self.rooms) self.passive_income_rate = data.get( "passive_income_rate", 0 )
