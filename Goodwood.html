# core/audio.py # Sound effects and audio management for CATINO. import pygame import numpy as np class AudioManager: def __init__(self): self.enabled = True self.volume = 0.4 self.music_mood = None self.current_music = None self.sfx = {} try: if not pygame.mixer.get_init(): pygame.mixer.init( frequency=22050, size=-16, channels=2, buffer=512 ) self._generate_sfx() except pygame.error as error: print(f"Audio initialization error: {error}") self.enabled = False def _generate_sfx(self): def tone(freq, duration, vol=0.3, wave="square"): sample_rate = 22050 n = int(sample_rate * duration) t = np.linspace(0, duration, n, False) if wave == "square": wave_data = np.sign( np.sin(2 * np.pi * freq * t) ) elif wave == "saw": wave_data = 2 * ( t * freq - np.floor(0.5 + t * freq) ) else: wave_data = np.sin(2 * np.pi * freq * t) envelope = np.linspace(1, 0, n) wave_data = wave_data * envelope * vol stereo = np.column_stack( (wave_data, wave_data) ) stereo = (stereo * 32767).astype(np.int16) return pygame.sndarray.make_sound(stereo) self.sfx["click"] = tone(600, 0.05, 0.2) self.sfx["spin"] = tone(200, 0.08, 0.15, "saw") self.sfx["win"] = tone(520, 0.12, 0.25) self.sfx["win2"] = tone(780, 0.15, 0.25) self.sfx["lose"] = tone(140, 0.2, 0.2, "saw") self.sfx["jackpot"] = tone(880, 0.3, 0.3) self.sfx["feed"] = tone(400, 0.1, 0.2) self.sfx["unlock"] = tone(660, 0.2, 0.3) def play_sfx(self, name: str): if not self.enabled: return sound = self.sfx.get(name) if sound: sound.set_volume(self.volume) sound.play() def set_volume(self, volume: float): self.volume = max(0.0, min(1.0, volume)) def set_enabled(self, enabled: bool): self.enabled = enabled if not enabled: self.stop_all() def set_mood(self, mood: str): if mood == self.music_mood: return self.music_mood = mood def stop_all(self): if pygame.mixer.get_init(): pygame.mixer.stop()
