# systems/collection.py # Manages owned cats, unlocking, friendship. from entities.cat import Cat from data.cats import get_all_cats, get_cat_data class CatCollection: def __init__(self): self.cats = {} # Create all cats as locked by default for cat_id in get_all_cats(): self.cats[cat_id] = Cat(cat_id, owned=False) # Starter self.unlock_cat("ryzhik") def unlock_cat(self, cat_id: str) -> bool: if cat_id not in self.cats: return False cat = self.cats[cat_id] if cat.owned: return False cat.owned = True cat.friendship = 5 return True def try_buy_cat(self, cat_id: str, resources) -> tuple[bool, str]: cat = self.cats.get(cat_id) if not cat: return False, "Нет такого кота" if cat.owned: return False, "Уже есть" cost = cat.unlock_cost if not resources.spend_coins(cost): return False, f"Нужно {cost} монет" self.unlock_cat(cat_id) resources.inc_stat("cats_unlocked") return True, f"Добро пожаловать, {cat.name}!" def feed_cat(self, cat_id: str, resources) -> tuple[bool, str]: cat = self.cats.get(cat_id) if not cat or not cat.owned: return False, "Нет кота" if not resources.spend_fish(1): return False, "Нет рыбок" leveled = cat.add_friendship(8 + cat.level * 2) msg = f"{cat.name} покушал! Дружба: {int(cat.friendship)}" if leveled: msg += " ★ Новый уровень дружбы!" return True, msg def get_owned(self) -> list: return [ c for c in self.cats.values() if c.owned ] def get_active_abilities(self) -> dict: """Aggregate all ability effects from owned cats with enough friendship.""" total = {} for cat in self.get_owned(): eff = cat.get_ability_effect() for k, v in eff.items(): if k == "visual_fx": total[k] = v elif isinstance(v, (int, float)): total[k] = total.get(k, 0) + v else: total[k] = v return total def to_dict(self) -> dict: return { cid: c.to_dict() for cid, c in self.cats.items() } def load_dict(self, data: dict): for cid, cdata in data.items(): if cid in self.cats: self.cats[cid] = Cat.from_dict(cdata)
