from data.achievements import ACHIEVEMENTS, get_achievement from data.cats import Rarity, get_all_cats class AchievementSystem: def __init__(self, resource_manager, cat_collection, empire): self.resources = resource_manager self.collection = cat_collection self.empire = empire self.unlocked = set() # ids self.just_unlocked = [] # for UI popup this frame def check_all(self): """Call periodically or after important events.""" self.just_unlocked.clear() for aid, ach in ACHIEVEMENTS.items(): if aid in self.unlocked: continue if self._check_condition(ach["condition"]): self._unlock(aid, ach) def _check_condition(self, cond: dict) -> bool: t = cond.get("type") if t == "stat": val = self.resources.get_stat(cond["key"], 0) op = cond.get("op", ">=") target = cond["value"] if op == ">=": return val >= target if op == ">": return val > target if op == "==": return val == target return False if t == "collection_count": owned = sum(1 for c in self.collection.cats.values() if c.owned) return owned >= cond["value"] if t == "rarity_complete": rarity_name = cond["rarity"] rarity = Rarity[rarity_name] all_of = [c for c in get_all_cats().values() if c["rarity"] == rarity] owned_of = [c for c in self.collection.cats.values() if c.owned and c.rarity == rarity] return len(owned_of) >= len(all_of) if t == "max_friendship": max_f = max((c.friendship for c in self.collection.cats.values() if c.owned), default=0) return max_f >= cond["value"] if t == "empire_level": return self.empire.level >= cond["value"] if t == "secret": return self.resources.get_stat(cond["key"], False) return False def _unlock(self, aid: str, ach: dict): self.unlocked.add(aid) self.just_unlocked.append(ach) # Give rewards reward = ach.get("reward", {}) if "coins" in reward: self.resources.add_coins(reward["coins"], "achievement") if "friendship_all" in reward: for cat in self.collection.cats.values(): if cat.owned: cat.add_friendship(reward["friendship_all"]) if "unlock_cat" in reward: self.collection.unlock_cat(reward["unlock_cat"]) def is_unlocked(self, aid: str) -> bool: return aid in self.unlocked def get_progress_text(self, aid: str) -> str: ach = get_achievement(aid) if not ach: return "" cond = ach["condition"] if cond["type"] == "stat": current = self.resources.get_stat(cond["key"], 0) return f"{current}/{cond['value']}" return "" def to_dict(self) -> dict: return {"unlocked": list(self.unlocked)} def load_dict(self, data: dict): self.unlocked = set(data.get("unlocked", []))
