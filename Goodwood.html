# systems/achievement_system.py # Checks conditions against current game state. from data.achievements import ACHIEVEMENTS, get_achievement from data.cats import Rarity, get_all_cats class AchievementSystem: def __init__(self, resource_manager, cat_collection, empire): self.resources = resource_manager self.collection = cat_collection self.empire = empire self.unlocked = set() self.just_unlocked = [] def check_all(self): """Call periodically or after important events.""" self.just_unlocked.clear() for aid, ach in ACHIEVEMENTS.items(): if aid in self.unlocked: continue if self._check_condition(ach["condition"]): self._unlock(aid, ach) def _check_condition(self, cond: dict) -> bool: condition_type = cond.get("type") if condition_type == "stat": value = self.resources.get_stat( cond["key"], 0 ) operator = cond.get("op", ">=") target = cond["value"] if operator == ">=": return value >= target if operator == ">": return value > target if operator == "==": return value == target return False if condition_type == "collection_count": owned = sum( 1 for cat in self.collection.cats.values() if cat.owned ) return owned >= cond["value"] if condition_type == "rarity_complete": rarity_name = cond["rarity"] rarity = Rarity[rarity_name] all_of = [ cat for cat in get_all_cats().values() if cat["rarity"] == rarity ] owned_of = [ cat for cat in self.collection.cats.values() if cat.owned and cat.rarity == rarity ] return len(owned_of) >= len(all_of) if condition_type == "max_friendship": max_friendship = max( ( cat.friendship for cat in self.collection.cats.values() if cat.owned ), default=0 ) return max_friendship >= cond["value"] if condition_type == "empire_level": return self.empire.level >= cond["value"] if condition_type == "secret": return self.resources.get_stat( cond["key"], False ) return False def _unlock(self, aid: str, ach: dict): self.unlocked.add(aid) self.just_unlocked.append(ach) # Give rewards reward = ach.get("reward", {}) if "coins" in reward: self.resources.add_coins( reward["coins"], "achievement" ) if "friendship_all" in reward: for cat in self.collection.cats.values(): if cat.owned: cat.add_friendship( reward["friendship_all"] ) if "unlock_cat" in reward: self.collection.unlock_cat( reward["unlock_cat"] ) def is_unlocked(self, aid: str) -> bool: return aid in self.unlocked def get_progress_text(self, aid: str) -> str: achievement = get_achievement(aid) if not achievement: return "" condition = achievement["condition"] if condition["type"] == "stat": current = self.resources.get_stat( condition["key"], 0 ) return f"{current}/{condition['value']}" return "" def to_dict(self) -> dict: return { "unlocked": list(self.unlocked) } def load_dict(self, data: dict): self.unlocked = set( data.get("unlocked", []) )
