already resolved inside pass elif self.state == GameState.ROULETTE: self.roulette.update(self.dt) elif self.state == GameState.REACTION: # mouse control mx, _ = pygame.mouse.get_pos() self.reaction.update(self.dt, mx / 960) # Passive income every ~10 sec # (simple) # ---------- INPUT ---------- def handle_event(self, event): if event.type == pygame.QUIT: self.running = False self.save() return if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: if self.state != GameState.HUB: self.state = GameState.HUB else: self.running = False self.save() if event.key == pygame.K_s: self.save() self.show_msg("Игра сохранена") if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1: self._handle_click(event.pos) def _handle_click(self, pos): x, y = pos self.audio.play_sfx("click") if self.state == GameState.HUB: #
